// src/components/Biblioteca/AgeRecommendations.astro
import libraryData from '../../data/processed/library.json';

const { ageGroups, books } = libraryData;

// Función para encontrar libro por título aproximado
const findBookByTitle = (title: string) => {
  return books.find(book => 
    book.title.toLowerCase().includes(title.toLowerCase()) ||
    title.toLowerCase().includes(book.title.toLowerCase())
  );
};

<section class="age-recommendations" id="recomendaciones-edad">
  <div class="section-header">
    <h2 class="title">
      <span class="icon">🌳</span>
      Árbol de Recomendaciones
    </h2>
    <p class="subtitle">Libros para cada etapa del crecimiento</p>
  </div>

  <div class="age-tree">
    <!-- Tronco del árbol (línea central) -->
    <div class="tree-trunk"></div>
    
    {ageGroups.map((group, index) => {
      const isEven = index % 2 === 0;
      const booksInCollection = group.booksInCollection.filter(b => b.inCollection);
      const booksToDiscover = group.booksInCollection.filter(b => !b.inCollection);
      
      return (
        <div class={`age-branch ${isEven ? 'left' : 'right'} branch-${index}`}>
          <!-- Rama del árbol -->
          <div class="branch-line"></div>
          
          <!-- Nodo de edad -->
          <div class="age-node">
            <span class="age-label">{group.age}</span>
          </div>
          
          <!-- Contenedor de libros -->
          <div class="books-container">
            <!-- Libros en la colección -->
            {booksInCollection.length > 0 && (
              <div class="books-section our-books">
                <h4 class="section-title">
                  <span class="icon">✅</span>
                  En nuestro estante
                </h4>
                <ul class="books-list">
                  {booksInCollection.slice(0, 3).map(book => {
                    const bookData = findBookByTitle(book.title);
                    return (
                      <li class="book-item">
                        {bookData ? (
                          <a 
                            href={`/biblioteca/libro/${bookData.slug}`}
                            class="book-link"
                            title={book.title}
                          >
                            {book.title}
                          </a>
                        ) : (
                          <span class="book-title">{book.title}</span>
                        )}
                      </li>
                    );
                  })}
                  {booksInCollection.length > 3 && (
                    <li class="more-books">
                      +{booksInCollection.length - 3} más
                    </li>
                  )}
                </ul>
              </div>
            )}
            
            <!-- Libros por descubrir -->
            {booksToDiscover.length > 0 && (
              <div class="books-section to-discover">
                <h4 class="section-title">
                  <span class="icon">🔍</span>
                  Por descubrir
                </h4>
                <ul class="books-list">
                  {booksToDiscover.slice(0, 3).map(book => (
                    <li class="book-item" title={book.title}>
                      {book.title}
                    </li>
                  ))}
                  {booksToDiscover.length > 3 && (
                    <li class="more-books">
                      +{booksToDiscover.length - 3} más
                    </li>
                  )}
                </ul>
              </div>
            )}
          </div>
        </div>
      );
    })}
  </div>

  <!-- Leyenda -->
  <div class="legend">
    <div class="legend-item">
      <span class="legend-icon">✅</span>
      <span>Ya en la biblioteca</span>
    </div>
    <div class="legend-item">
      <span class="legend-icon">🔍</span>
      <span>Por descubrir</span>
    </div>
  </div>
</section>

<script>
// Script para manejar el modo noche/día
if (typeof window !== 'undefined') {
  document.addEventListener('DOMContentLoaded', () => {
    const toggleButton = document.getElementById('night-mode-toggle');
    const htmlElement = document.documentElement;
    
    // Verificar preferencia guardada o del sistema
    const savedTheme = localStorage.getItem('theme');
    const systemPrefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
    
    // Aplicar tema inicial
    if (savedTheme === 'dark' || (!savedTheme && systemPrefersDark)) {
      htmlElement.setAttribute('data-theme', 'dark');
    } else {
      htmlElement.setAttribute('data-theme', 'light');
    }
    
    // Manejar clic en el toggle
    if (toggleButton) {
      toggleButton.addEventListener('click', () => {
        const currentTheme = htmlElement.getAttribute('data-theme');
        const newTheme = currentTheme === 'light' ? 'dark' : 'light';
        
        // Cambiar tema
        htmlElement.setAttribute('data-theme', newTheme);
        localStorage.setItem('theme', newTheme);
        
        // Disparar evento personalizado para que otros componentes reaccionen
        window.dispatchEvent(new CustomEvent('themechange', { 
          detail: { theme: newTheme } 
        }));
      });
    }
    
    // Escuchar cambios en las preferencias del sistema
    window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', (e) => {
      if (!localStorage.getItem('theme')) {
        const newTheme = e.matches ? 'dark' : 'light';
        htmlElement.setAttribute('data-theme', newTheme);
      }
    });
  });
}
</script>
